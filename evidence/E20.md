function chalsEnsureAuth_(settings) {
  // Silent auto-login: exchange the Google identity token this add-on already
  // holds (the user is signed in to Google Sheets) for a Chals workspace API
  // key, so the user never pastes a key. Falls back to manual entry if the
  // identity token is unavailable or the exchange fails.
  var manualNotice = 'Sign in with your Google account to start, or add a workspace API key in Settings.';
  var failNotice = 'Could not connect to Chals automatically. You can add a workspace API key in Settings.';
  var idToken = '';
  try {
    idToken = ScriptApp.getIdentityToken();
  } catch (err) {
    idToken = '';
  }
  if (!idToken) {
    return { settings: settings, notice: manualNotice };
  }
  // A brand-new account can be momentarily unreadable right after signup
  // (BigQuery streaming buffer) -> the backend returns a retryable 503. Retry a
  // few times so even the very first open lands silently.
  var result = null;
  for (var attempt = 0; attempt < CHALS_AUTH_PROVISION_MAX_ATTEMPTS; attempt += 1) {
    try {
      result = provisionChalsAuth_(settings.apiBase, idToken);
    } catch (err) {
      return { settings: settings, notice: failNotice };
    }
    if (result && result.apiKey) {
      break;
    }
    if (!(result && result.retryable) || attempt === CHALS_AUTH_PROVISION_MAX_ATTEMPTS - 1) {
      break;
    }
    Utilities.sleep(CHALS_AUTH_PROVISION_RETRY_MS);
  }
  if (!result || !result.apiKey) {
    return {
      settings: settings,
      notice: (result && result.error) ? result.error : failNotice
    };
  }
  var next = readChalsSettings_();
  next.apiKey = result.apiKey;
  next.updatedAt = new Date().toISOString();
  writeChalsSettings_(next);
  return { settings: next, notice: '' };
}
