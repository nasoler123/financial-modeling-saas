export const parseManualFormulaToAst = (input: string, target: FormulaTarget): FormulaAstNode => {
  const trimmed = input.trim();
  if (!trimmed) {
    throw new Error(t('The formula cannot be empty.'));
  }
  const tokens = tokenizeManualFormula(trimmed);
  if (!tokens.length) {
    throw new Error(t('The formula cannot be empty.'));
  }
  const parser = new Parser(tokens, target);
  return parser.parse();
};

export const buildReferenceToken = (
  driverId: string,
  mode: 'forecast' | 'actual' | `actual-${number}` | `actual@${string}`
): string => `@${driverId}[${mode}]`;
