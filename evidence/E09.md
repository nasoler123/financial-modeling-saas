def _tarjan_scc(
    nodes: List[str],
    *,
    neighbors: Callable[[str], List[str]],
) -> List[List[str]]:
    """
    Tarjan SCC. Deterministic if `nodes` and `neighbors(v)` are sorted.
    Returns list of components (each component is a sorted list of node ids).
    """
    index = 0
    stack: List[str] = []
    on_stack: Set[str] = set()
    idx: Dict[str, int] = {}
    low: Dict[str, int] = {}
    out: List[List[str]] = []

    def strongconnect(v: str) -> None:
        nonlocal index
        idx[v] = index
        low[v] = index
        index += 1
        stack.append(v)
        on_stack.add(v)

        for w in neighbors(v):
            if w not in idx:
                strongconnect(w)
                low[v] = min(low[v], low[w])
            elif w in on_stack:
                low[v] = min(low[v], idx[w])

        if low[v] == idx[v]:
            comp: List[str] = []
            while True:
                w = stack.pop()
                on_stack.remove(w)
                comp.append(w)
                if w == v:
                    break
            comp.sort()
            out.append(comp)

    for v in nodes:
        if v not in idx:
            strongconnect(v)

    return out
